from pathlib import Path                    # 导入 Path，用于安全拼接文件路径

"""FastAPI + SQLAlchemy 数据库连接层。"""  # 模块说明

from sqlalchemy import create_engine          # 创建数据库引擎
from sqlalchemy.orm import DeclarativeBase, sessionmaker  # ORM 基类和会话工厂

from app.config import SQLALCHEMY_DATABASE_URL  # 读取数据库连接地址

connect_args = (                       # SQLite 专用连接参数
    {"check_same_thread": False}       # 允许 FastAPI 多线程访问同一个 SQLite
    if SQLALCHEMY_DATABASE_URL.startswith("sqlite")  # 只有 SQLite 才需要
    else {}                            # 其他数据库不需要
)
engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args=connect_args)  # 创建引擎

SessionLocal = sessionmaker(bind=engine)  # 创建会话工厂，每次调用生成一个新 Session


class Base(DeclarativeBase):  # 所有 ORM 模型都要继承的基类
    pass                      # 不需要额外实现


def get_db():                 # FastAPI 依赖：提供数据库 Session
    db = SessionLocal()       # 创建 Session
    try:                      # 开始 try
        yield db              # 把 Session 交给路由使用
    finally:                  # 无论成功还是异常
        db.close()            # 请求结束后关闭 Session
```

"""第 6 周 - 第 3 步：FastAPI + SQLAlchemy ORM 模型"""  # 模块说明

from app.database import Base                  # 导入 ORM 基类
from sqlalchemy.orm import Mapped, mapped_column, relationship  # 导入 ORM 类型和关系
from typing import List, Optional             # 类型提示：列表和可选
from datetime import datetime                 # 时间类型
from sqlalchemy import DateTime, Float, ForeignKey, Integer, String  # 数据库列类型


# 用户表
class User(Base):                 # 用户 ORM 模型
    __tablename__ = "users"       # 对应数据库表名

    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)  # 主键，自增
    username: Mapped[str] = mapped_column(String(50), unique=True, index=True, nullable=False)  # 用户名，唯一且索引
    password: Mapped[str] = mapped_column(String(100), nullable=False)  # 密码哈希
    age: Mapped[int] = mapped_column(Integer, default=18)  # 年龄，默认 18
    email: Mapped[Optional[str]] = mapped_column(String(120), nullable=True)  # 邮箱，可空
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)  # 创建时间

    # 一个用户拥有多个商品
    items: Mapped[List["Item"]] = relationship(back_populates="user")  # 一对多关系

    def __repr__(self):  # 便于调试的输出
        return f"<User(id={self.id}, username={self.username!r})>"  # 返回可读表示


# 商品表
class Item(Base):                  # 商品 ORM 模型
    __tablename__ = "items"        # 对应数据库表名

    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)  # 主键，自增
    name: Mapped[str] = mapped_column(String(50), nullable=False)  # 商品名称
    price: Mapped[float] = mapped_column(Float, nullable=False)    # 商品价格
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id"), nullable=False)  # 外键指向用户
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)  # 创建时间

    # 商品属于哪个用户
    user: Mapped["User"] = relationship(back_populates="items")  # 反向关系

    def __repr__(self):  # 便于调试的输出
        return f"<Item(id={self.id}, name={self.name!r})>"  # 返回可读表示
```

## 6. app/security.py：密码哈希与 JWT

```python
"""密码哈希与 JWT 工具。"""  # 模块说明

from datetime import datetime, timedelta, timezone  # 时间和时区处理

import bcrypt  # bcrypt 密码哈希库
import jwt     # PyJWT 库，用于签发/解析 token

from app.config import ACCESS_TOKEN_EXPIRE_MINUTES, ALGORITHM, SECRET_KEY  # 读取配置


def hash_password(password: str) -> str:          # 生成密码哈希
    """注册时把明文密码转成 bcrypt 哈希字符串。"""  # 函数说明
    hashed = bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt())  # 加盐哈希
    return hashed.decode("utf-8")                 # 返回字符串形式


def verify_password(password: str, hashed_password: str) -> bool:  # 校验密码
    """登录时校验明文密码和数据库里的哈希是否匹配。"""  # 函数说明
    return bcrypt.checkpw(password.encode("utf-8"), hashed_password.encode("utf-8"))  # 返回是否匹配


def create_access_token(user_id: int) -> str:     # 签发 JWT
    """根据用户 id 签发 JWT。"""                   # 函数说明
    expire = datetime.now(timezone.utc) + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)  # 计算过期时间
    payload = {"sub": str(user_id), "exp": expire}  # sub 存用户 id，exp 存过期时间
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)  # 用密钥签名


def decode_access_token(token: str) -> int:       # 解析 JWT
    """解析 JWT，返回用户 id。"""                  # 函数说明
    payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])  # 验签并解码
    return int(payload["sub"])                    # 返回用户 id
```

```python
"""统一异常处理。"""  # 模块说明

import logging  # 日志模块

from fastapi import FastAPI, Request  # FastAPI 应用和请求对象
from fastapi.encoders import jsonable_encoder  # 把对象转成 JSON 可序列化数据
from fastapi.exceptions import RequestValidationError  # 参数校验错误
from fastapi.responses import JSONResponse  # JSON 响应
from starlette.exceptions import HTTPException as StarletteHTTPException  # Starlette 的 HTTP 异常

from app.exceptions import BusinessError  # 自定义业务异常

logger = logging.getLogger(__name__)  # 创建当前模块的日志器


def register_exception_handlers(app: FastAPI) -> None:  # 注册所有异常处理器
    @app.exception_handler(BusinessError)  # 处理自定义业务异常
    async def business_error_handler(request: Request, exc: BusinessError):  # 处理器函数
        return JSONResponse(  # 返回 JSON 错误
            status_code=exc.status_code,  # 使用异常携带的状态码
            content={  # 响应体
                "code": exc.code,    # 错误码
                "message": exc.message,  # 错误信息
            },
        )

    @app.exception_handler(StarletteHTTPException)  # 处理 HTTPException
    async def http_error_handler(request: Request, exc: StarletteHTTPException):  # 处理器
        return JSONResponse(  # 返回 JSON
            status_code=exc.status_code,  # HTTP 状态码
            content={  # 响应体
                "code": "HTTP_ERROR",  # 统一错误码
                "message": str(exc.detail),  # 错误详情
            },
        )

    @app.exception_handler(RequestValidationError)  # 处理参数校验错误
    async def validation_error_handler(request: Request, exc: RequestValidationError):  # 处理器
        return JSONResponse(  # 返回 JSON
            status_code=422,  # 固定 422
            content={  # 响应体
                "code": "VALIDATION_ERROR",  # 错误码
                "message": "参数校验失败",     # 提示
                "detail": jsonable_encoder(exc.errors()),  # 详细错误列表
            },
        )

    @app.exception_handler(Exception)  # 兜底处理所有未捕获异常
    async def unhandled_exception_handler(request: Request, exc: Exception):  # 处理器
        logger.exception("Unhandled error: %s", exc)  # 记录完整异常日志
        return JSONResponse(  # 返回 JSON
            status_code=500,  # 固定 500
            content={  # 响应体
                "code": "INTERNAL_ERROR",  # 错误码
                "message": "服务器内部错误",  # 不泄露内部细节
            },
        )
```

## 9. app/middleware.py：中间件

```python
"""FastAPI 中间件注册。"""  # 模块说明

import logging  # 日志模块
import time     # 计时模块
import uuid     # 生成 Request-ID

from fastapi import Request  # 请求对象
from fastapi.middleware.cors import CORSMiddleware  # CORS 中间件
from starlette.middleware.base import BaseHTTPMiddleware  # 自定义中间件基类

from app.config import CORS_ORIGINS  # 允许跨域的来源

logger = logging.getLogger(__name__)  # 当前模块日志器


class RequestIDMiddleware(BaseHTTPMiddleware):  # Request-ID 中间件
    async def dispatch(self, request: Request, call_next):  # 处理每个请求
        request_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())  # 用传入 ID 或生成新 ID
        request.state.request_id = request_id  # 把 ID 存到请求状态，方便日志使用
        response = await call_next(request)    # 交给下一个中间件或路由
        response.headers["X-Request-ID"] = request_id  # 响应头带上 Request-ID
        return response  # 返回响应


class LogMiddleware(BaseHTTPMiddleware):  # 日志中间件
    async def dispatch(self, request: Request, call_next):  # 处理每个请求
        start = time.time()  # 记录开始时间
        response = await call_next(request)  # 执行后续处理
        cost = round((time.time() - start) * 1000, 2)  # 计算耗时（毫秒）
        response.headers["X-Process-Time"] = str(cost)  # 响应头返回耗时
        request_id = getattr(request.state, "request_id", "-")  # 读取 Request-ID
        logger.info(  # 输出访问日志
            "[%s] %s %s 耗时 %.2fms",  # 日志格式
            request_id,                # Request-ID
            request.method,            # 请求方法
            request.url.path,          # 请求路径
            cost,                      # 耗时
        )
        return response  # 返回响应


def register_middlewares(app):  # 注册所有中间件
    app.add_middleware(  # 添加 CORS
        CORSMiddleware,  # 中间件类型
        allow_origins=CORS_ORIGINS,  # 允许的来源
        allow_credentials=True,      # 允许携带凭证
        allow_methods=["*"],         # 允许所有方法
        allow_headers=["*"],         # 允许所有请求头
    )
    app.add_middleware(RequestIDMiddleware)  # 添加 Request-ID
    app.add_middleware(LogMiddleware)        # 添加日志
```

## 10. app/dependencies.py：依赖注入

```python
import jwt  # PyJWT 库，用于捕获 token 解析错误
from fastapi import Depends, HTTPException  # 依赖注入和 HTTP 异常
from fastapi.security import OAuth2PasswordBearer  # Bearer token 安全方案
from sqlalchemy.orm import Session  # 数据库 Session 类型

from app.database import get_db  # 数据库依赖
from app.models import User      # 用户模型
from app.security import decode_access_token  # 解析 token 的函数

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="auth/login")  # 声明登录地址，FastAPI 从 Authorization 头取 token


class Pagination:  # 分页依赖
    def __init__(self, page: int = 1, size: int = 10):  # 默认页码和每页数量
        self.page = max(page, 1)            # 页码最小为 1
        self.size = min(max(size, 1), 100)  # 每页数量限制在 1-100


def get_current_user(  # 获取当前登录用户
    token: str = Depends(oauth2_scheme),  # 从请求头拿 token
    db: Session = Depends(get_db),        # 获取数据库 Session
) -> User:  # 返回 User 对象
    try:  # 尝试解析
        user_id = decode_access_token(token)  # 解析 token 得到用户 id
    except (jwt.PyJWTError, KeyError, TypeError, ValueError):  # 捕获各种解析错误
        raise HTTPException(401, "无效的认证凭证")  # 返回 401
    user = db.get(User, user_id)  # 按主键查用户
    if user is None:  # 用户不存在
        raise HTTPException(401, "用户不存在")  # 返回 401
    return user  # 返回当前用户


def require_login(user: User = Depends(get_current_user)):  # 登录快捷依赖
    return user  # 直接返回已认证用户
```

## 11. app/routers/auth.py：认证路由

```python
"""认证路由。"""  # 模块说明

from fastapi import APIRouter, Depends, HTTPException  # 路由、依赖、异常
from fastapi.security import OAuth2PasswordRequestForm  # 表单登录字段
from sqlalchemy.orm import Session  # 数据库 Session

from app.database import get_db  # 数据库依赖
from app.models import User      # 用户模型
from app.schemas import Token    # 登录响应模型
from app.security import create_access_token, verify_password  # 签发 token 和校验密码

router = APIRouter(prefix="/auth", tags=["认证"])  # 创建认证路由，统一前缀 /auth


@router.post(  # 注册 POST 路由
    "/login",                 # 登录路径
    response_model=Token,     # 响应模型
    summary="登录并获取JWT",   # Swagger 标题
    description="使用表单提交 username 和 password；密码正确返回 access_token。",  # 说明
)
def login(  # 登录处理函数
    form_data: OAuth2PasswordRequestForm = Depends(),  # 自动解析表单 username/password
    db: Session = Depends(get_db),                     # 数据库依赖
):
    """校验密码并签发 JWT。"""  # 函数说明
    user = db.query(User).filter(User.username == form_data.username).first()  # 按用户名查用户
    if user is None or not verify_password(form_data.password, user.password):  # 用户不存在或密码错误
        raise HTTPException(status_code=401, detail="用户名或密码错误")  # 返回 401
    token = create_access_token(user.id)  # 签发 token
    return Token(access_token=token)      # 返回 token
```

def register(user: UserRegister, db: Session = Depends(get_db)):  # 注册函数
    """注册新用户：查重后写入密码哈希。"""  # 函数说明
    existing = db.query(User).filter(User.username == user.username).first()  # 查重
    if existing:  # 用户名已存在
        raise HTTPException(status_code=400, detail="用户名已存在")  # 返回 400
    db_user = User(  # 创建用户
        username=user.username,          # 用户名
        password=hash_password(user.password),  # 密码哈希
        age=user.age,                    # 年龄
        email=user.email,                # 邮箱
    )
    db.add(db_user)      # 加入会话
    db.commit()          # 提交事务
    db.refresh(db_user)  # 刷新数据库生成的值
    return db_user       # 返回用户

    return db.query(User).offset(offset).limit(pagination.size).all()  # 分页查询

def read_me(current: User = Depends(require_login)):  # 当前用户接口

```python
"""商品路由。"""  # 模块说明

from fastapi import APIRouter, Depends, HTTPException, Query, status  # 路由、依赖、异常、查询参数、状态码
from sqlalchemy.orm import Session  # 数据库 Session

from app.database import get_db  # 数据库依赖

router = APIRouter(tags=["商品"])  # 创建商品路由


@router.get(  # 注册 GET 路由
    "/news",  # 路径
    summary="新闻列表（查询参数示例）",  # Swagger 标题
    description="演示 Query 查询参数：skip 跳过的记录数，limit 返回的记录数。",  # 说明
)
async def get_news(  # 查询参数示例
    skip: int = Query(0, description="跳过的记录数", ge=0, lt=100),  # skip 参数
    limit: int = Query(10, description="返回的记录数", ge=1, le=100),  # limit 参数
):
    """返回查询参数示例。"""  # 函数说明
    return {"skip": skip, "limit": limit}  # 原样返回参数


@router.get(  # 注册 GET 路由
    "/items",                     # 商品列表路径
    response_model=list[ItemOut], # 响应模型
    summary="商品分页列表",        # Swagger 标题
    description="使用分页依赖 Pagination，按 page 和 size 返回商品。",  # 说明
)
def list_items(pagination: Pagination = Depends(), db: Session = Depends(get_db)):  # 商品列表
    """分页查询商品列表。"""  # 函数说明
    offset = (pagination.page - 1) * pagination.size  # 计算偏移
    return db.query(Item).offset(offset).limit(pagination.size).all()  # 分页查询


@router.post(  # 注册 POST 路由
    "/items",  # 创建商品路径
    response_model=ItemOut,  # 响应模型
    status_code=status.HTTP_201_CREATED,  # 201 创建成功
    summary="创建商品",  # Swagger 标题
    description="需要登录；商品归属当前登录用户。",  # 说明
)
def create_item(  # 创建商品
    item: ItemCreate,                    # 请求体
    current: User = Depends(require_login),  # 当前登录用户
    db: Session = Depends(get_db),       # 数据库依赖
):
    """创建当前用户的商品。"""  # 函数说明
    db_item = Item(**item.model_dump(), user_id=current.id)  # 创建商品并绑定当前用户
    db.add(db_item)      # 加入会话
    db.commit()          # 提交
    db.refresh(db_item)  # 刷新
    return db_item       # 返回


@router.get(  # 注册 GET 路由
    "/items/{item_id}",  # 商品 id 路径
    response_model=ItemOut,  # 响应模型
    summary="查询单个商品",  # Swagger 标题
    description="按主键查询商品；不存在返回 404。",  # 说明
)
def read_item(item_id: int, db: Session = Depends(get_db)):  # 查询单个商品
    """按主键查询商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 按主键查询
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 返回 404
    return item  # 返回商品


@router.put(  # 注册 PUT 路由
    "/items/{item_id}",  # 更新路径
    response_model=ItemOut,  # 响应模型
    summary="更新商品",  # Swagger 标题
    description="只能更新自己创建的商品。",  # 说明
)
def update_item(  # 更新商品
    item_id: int,                            # 商品 id
    payload: ItemUpdate,                     # 部分更新请求体
    current: User = Depends(require_login),  # 当前用户
    db: Session = Depends(get_db),           # 数据库
):
    """部分更新自己的商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 查询商品
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 404
    if item.user_id != current.id:  # 不是本人
        raise HTTPException(status_code=403, detail="无权操作该商品")  # 403
    for field, value in payload.model_dump(exclude_unset=True).items():  # 只更新传过的字段
        setattr(item, field, value)  # 设置新值
    db.commit()          # 提交
    db.refresh(item)     # 刷新
    return item          # 返回


@router.delete(  # 注册 DELETE 路由
    "/items/{item_id}",  # 删除路径
    status_code=status.HTTP_204_NO_CONTENT,  # 204 无内容
    summary="删除商品",  # Swagger 标题
    description="只能删除自己创建的商品。",  # 说明
)
def delete_item(  # 删除商品
    item_id: int,                            # 商品 id
    current: User = Depends(require_login),  # 当前用户
    db: Session = Depends(get_db),           # 数据库
):
    """删除自己的商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 查询商品
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 404
    if item.user_id != current.id:  # 不是本人
        raise HTTPException(status_code=403, detail="无权操作该商品")  # 403
    db.delete(item)   # 删除
    db.commit()       # 提交
```

from pathlib import Path  # 路径处理

from fastapi import APIRouter, BackgroundTasks, File, Form, UploadFile  # 路由、后台任务、表单、文件

    return RedirectResponse("/")  # 跳转到首页

    "/send",  # 路径

async def send_notice(  # 后台任务
