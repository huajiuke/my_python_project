```python
from pydantic import BaseModel

class Author(BaseModel):
    name: str
    country: str

class BookWithAuthor(BaseModel):
    title: str
    author: Author

data = {
    "title": "Python入门",
    "author": {"name": "张三", "country": "中国"},
}
book = BookWithAuthor.model_validate(data)
```
