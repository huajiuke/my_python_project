```python
from pathlib import Path  # 路径处理

from fastapi import APIRouter, BackgroundTasks, File, Form, UploadFile  # 路由、后台任务、表单、文件
from fastapi.responses import FileResponse, HTMLResponse, JSONResponse, RedirectResponse  # 各种响应类型

router = APIRouter(tags=["文件与响应"])  # 创建路由
STATIC_DIR = Path(__file__).resolve().parent.parent / "static"  # 计算静态目录


@router.get(  # 注册 GET 路由
    "/json",  # 路径
    summary="返回JSON",  # Swagger 标题
    description="演示 JSONResponse 返回 JSON 数据。",  # 说明
)
async def get_json():  # 返回 JSON
    """返回 JSON 响应。"""  # 函数说明
    return JSONResponse({"msg": "这是JSON"})  # 显式 JSON 响应


@router.get(  # 注册 GET 路由
    "/html",  # 路径
    summary="返回HTML",  # Swagger 标题
    description="演示 HTMLResponse 返回 HTML 页面。",  # 说明
)
async def get_html():  # 返回 HTML
    """返回 HTML 响应。"""  # 函数说明
    return HTMLResponse("<h1>Hello FastAPI</h1>")  # 返回 HTML


@router.get(  # 注册 GET 路由
    "/file",  # 路径
    summary="返回文件",  # Swagger 标题
    description="演示 FileResponse，返回 static/hello.html 文件。",  # 说明
)
async def get_file():  # 返回文件
    """返回静态文件。"""  # 函数说明
    return FileResponse(STATIC_DIR / "hello.html")  # 返回文件响应


@router.get(  # 注册 GET 路由
    "/redirect",  # 路径
    summary="重定向",  # Swagger 标题
    description="演示 RedirectResponse，跳转到首页 /。",  # 说明
)
async def redirect():  # 重定向
    """重定向到首页。"""  # 函数说明
    return RedirectResponse("/")  # 跳转到首页


def send_email(to: str):  # 模拟发送邮件
    print(f"send email to {to}")  # 打印日志


@router.post(  # 注册 POST 路由
    "/send",  # 路径
    summary="发送通知（后台任务）",  # Swagger 标题
    description="接收 email 表单字段，响应后执行后台邮件任务。",  # 说明
)
async def send_notice(  # 后台任务
    background_tasks: BackgroundTasks,  # FastAPI 自动注入后台任务
    email: str = Form(...),             # 表单必填字段
):
    """把发邮件加入后台任务，立即返回受理结果。"""  # 函数说明
    background_tasks.add_task(send_email, email)  # 添加后台任务
    return {"message": "已受理"}  # 立即返回


@router.post(  # 注册 POST 路由
    "/upload",  # 路径
    summary="上传文件",  # Swagger 标题
    description="接收 username 表单字段和 file 文件，返回文件信息。",  # 说明
)
async def upload(username: str = Form(...), file: UploadFile = File(...)):  # 上传文件
    """读取上传文件并返回文件名和大小。"""  # 函数说明
    content = await file.read()  # 读取文件内容
    return {"username": username, "filename": file.filename, "size": len(content)}  # 返回文件信息
```
