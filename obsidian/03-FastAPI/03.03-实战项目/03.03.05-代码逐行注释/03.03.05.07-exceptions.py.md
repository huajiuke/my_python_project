```python
"""业务异常定义。"""  # 模块说明


class BusinessError(Exception):  # 自定义业务异常，继承 Exception
    """业务异常：status_code 对应 HTTP 状态码，message 是用户可见信息。"""  # 类说明

    def __init__(  # 构造函数
        self,      # 实例自身
        message: str,              # 用户可见错误信息
        status_code: int = 400,    # HTTP 状态码，默认 400
        code: str = "BUSINESS_ERROR",  # 错误码，前端可用来判断
    ):
        super().__init__(message)  # 调用 Exception 初始化
        self.message = message     # 保存 message
        self.status_code = status_code  # 保存状态码
        self.code = code           # 保存错误码
```
