```python
"""项目配置：优先从环境变量读取，未设置时使用开发默认值。"""  # 模块说明

import os              # 导入 os，用于读取环境变量
from pathlib import Path  # 导入 Path，用于处理路径

BASE_DIR = Path(__file__).resolve().parent.parent  # 得到项目根目录 fastApiProject
DATA_DIR = BASE_DIR / "app" / "data"               # 数据库文件目录
DATA_DIR.mkdir(parents=True, exist_ok=True)        # 目录不存在时自动创建
DATABASE_PATH = DATA_DIR / "fastapi.db"            # 默认 SQLite 数据库路径

SQLALCHEMY_DATABASE_URL = os.getenv(   # 读取数据库连接地址
    "DATABASE_URL",                    # 环境变量名
    f"sqlite:///{DATABASE_PATH.as_posix()}",  # 默认值
)
SECRET_KEY = os.getenv(                # 读取 JWT 签名密钥
    "SECRET_KEY",                      # 环境变量名
    "dev-secret-change-me-please-use-env-1234567890",  # 开发默认值
)
ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")  # JWT 签名算法
ACCESS_TOKEN_EXPIRE_MINUTES = int(os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "30"))  # token 有效期
CORS_ORIGINS = [                       # 允许跨域的来源列表
    origin.strip()                     # 去掉空格
    for origin in os.getenv("CORS_ORIGINS", "http://localhost:3000").split(",")  # 逗号分隔
    if origin.strip()                  # 过滤空值
]
```
