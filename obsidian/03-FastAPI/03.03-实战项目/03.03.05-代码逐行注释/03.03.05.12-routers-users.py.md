```python
"""用户路由。"""  # 模块说明

from fastapi import APIRouter, Depends, HTTPException  # 路由、依赖、异常
from sqlalchemy.orm import Session  # 数据库 Session

from app.database import get_db  # 数据库依赖
from app.dependencies import Pagination, require_login  # 分页和登录依赖
from app.models import User      # 用户模型
from app.schemas import UserOut, UserRegister  # 用户请求/响应模型
from app.security import hash_password  # 密码哈希函数

router = APIRouter(prefix="/users", tags=["用户"])  # 创建用户路由


@router.post(  # 注册 POST 路由
    "/register",          # 注册路径
    response_model=UserOut,  # 响应模型
    status_code=201,      # 创建成功状态码
    summary="用户注册",    # Swagger 标题
    description="创建新用户；用户名重复时返回 400。",  # 说明
)
def register(user: UserRegister, db: Session = Depends(get_db)):  # 注册函数
    """注册新用户：查重后写入密码哈希。"""  # 函数说明
    existing = db.query(User).filter(User.username == user.username).first()  # 查重
    if existing:  # 用户名已存在
        raise HTTPException(status_code=400, detail="用户名已存在")  # 返回 400
    db_user = User(  # 创建用户
        username=user.username,          # 用户名
        password=hash_password(user.password),  # 密码哈希
        age=user.age,                    # 年龄
        email=user.email,                # 邮箱
    )
    db.add(db_user)      # 加入会话
    db.commit()          # 提交事务
    db.refresh(db_user)  # 刷新数据库生成的值
    return db_user       # 返回用户


@router.get(  # 注册 GET 路由
    "",                     # 路径 /users
    response_model=list[UserOut],  # 返回用户列表
    summary="用户列表",      # Swagger 标题
    description="分页返回所有用户，不包含密码字段。",  # 说明
)
def list_users(pagination: Pagination = Depends(), db: Session = Depends(get_db)):  # 用户列表
    """分页查询全部用户。"""  # 函数说明
    offset = (pagination.page - 1) * pagination.size  # 计算偏移量
    return db.query(User).offset(offset).limit(pagination.size).all()  # 分页查询


@router.get(  # 注册 GET 路由
    "/me",                  # 当前用户路径
    response_model=UserOut, # 响应模型
    summary="当前登录用户",  # Swagger 标题
    description="解析 JWT 后返回当前用户；未登录返回 401。",  # 说明
)
def read_me(current: User = Depends(require_login)):  # 当前用户接口
    """返回当前登录用户。"""  # 函数说明
    return current  # 返回已认证用户


@router.get(  # 注册 GET 路由
    "/{user_id}",           # 路径参数
    response_model=UserOut, # 响应模型
    summary="按ID查询用户",  # Swagger 标题
    description="按主键查询单个用户；不存在返回 404。",  # 说明
)
def get_user(user_id: int, db: Session = Depends(get_db)):  # 按 id 查用户
    """按主键查询用户。"""  # 函数说明
    user = db.get(User, user_id)  # 按主键查询
    if not user:  # 不存在
        raise HTTPException(status_code=404, detail="用户不存在")  # 返回 404
    return user  # 返回用户
```
