```python
"""商品路由。"""  # 模块说明

from fastapi import APIRouter, Depends, HTTPException, Query, status  # 路由、依赖、异常、查询参数、状态码
from sqlalchemy.orm import Session  # 数据库 Session

from app.database import get_db  # 数据库依赖
from app.dependencies import Pagination, require_login  # 分页和登录依赖
from app.models import Item, User  # 商品和用户模型
from app.schemas import ItemCreate, ItemOut, ItemUpdate  # 商品请求/响应模型

router = APIRouter(tags=["商品"])  # 创建商品路由


@router.get(  # 注册 GET 路由
    "/news",  # 路径
    summary="新闻列表（查询参数示例）",  # Swagger 标题
    description="演示 Query 查询参数：skip 跳过的记录数，limit 返回的记录数。",  # 说明
)
async def get_news(  # 查询参数示例
    skip: int = Query(0, description="跳过的记录数", ge=0, lt=100),  # skip 参数
    limit: int = Query(10, description="返回的记录数", ge=1, le=100),  # limit 参数
):
    """返回查询参数示例。"""  # 函数说明
    return {"skip": skip, "limit": limit}  # 原样返回参数


@router.get(  # 注册 GET 路由
    "/items",                     # 商品列表路径
    response_model=list[ItemOut], # 响应模型
    summary="商品分页列表",        # Swagger 标题
    description="使用分页依赖 Pagination，按 page 和 size 返回商品。",  # 说明
)
def list_items(pagination: Pagination = Depends(), db: Session = Depends(get_db)):  # 商品列表
    """分页查询商品列表。"""  # 函数说明
    offset = (pagination.page - 1) * pagination.size  # 计算偏移
    return db.query(Item).offset(offset).limit(pagination.size).all()  # 分页查询


@router.post(  # 注册 POST 路由
    "/items",  # 创建商品路径
    response_model=ItemOut,  # 响应模型
    status_code=status.HTTP_201_CREATED,  # 201 创建成功
    summary="创建商品",  # Swagger 标题
    description="需要登录；商品归属当前登录用户。",  # 说明
)
def create_item(  # 创建商品
    item: ItemCreate,                    # 请求体
    current: User = Depends(require_login),  # 当前登录用户
    db: Session = Depends(get_db),       # 数据库依赖
):
    """创建当前用户的商品。"""  # 函数说明
    db_item = Item(**item.model_dump(), user_id=current.id)  # 创建商品并绑定当前用户
    db.add(db_item)      # 加入会话
    db.commit()          # 提交
    db.refresh(db_item)  # 刷新
    return db_item       # 返回


@router.get(  # 注册 GET 路由
    "/items/{item_id}",  # 商品 id 路径
    response_model=ItemOut,  # 响应模型
    summary="查询单个商品",  # Swagger 标题
    description="按主键查询商品；不存在返回 404。",  # 说明
)
def read_item(item_id: int, db: Session = Depends(get_db)):  # 查询单个商品
    """按主键查询商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 按主键查询
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 返回 404
    return item  # 返回商品


@router.put(  # 注册 PUT 路由
    "/items/{item_id}",  # 更新路径
    response_model=ItemOut,  # 响应模型
    summary="更新商品",  # Swagger 标题
    description="只能更新自己创建的商品。",  # 说明
)
def update_item(  # 更新商品
    item_id: int,                            # 商品 id
    payload: ItemUpdate,                     # 部分更新请求体
    current: User = Depends(require_login),  # 当前用户
    db: Session = Depends(get_db),           # 数据库
):
    """部分更新自己的商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 查询商品
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 404
    if item.user_id != current.id:  # 不是本人
        raise HTTPException(status_code=403, detail="无权操作该商品")  # 403
    for field, value in payload.model_dump(exclude_unset=True).items():  # 只更新传过的字段
        setattr(item, field, value)  # 设置新值
    db.commit()          # 提交
    db.refresh(item)     # 刷新
    return item          # 返回


@router.delete(  # 注册 DELETE 路由
    "/items/{item_id}",  # 删除路径
    status_code=status.HTTP_204_NO_CONTENT,  # 204 无内容
    summary="删除商品",  # Swagger 标题
    description="只能删除自己创建的商品。",  # 说明
)
def delete_item(  # 删除商品
    item_id: int,                            # 商品 id
    current: User = Depends(require_login),  # 当前用户
    db: Session = Depends(get_db),           # 数据库
):
    """删除自己的商品。"""  # 函数说明
    item = db.get(Item, item_id)  # 查询商品
    if item is None:  # 不存在
        raise HTTPException(status_code=404, detail="商品不存在")  # 404
    if item.user_id != current.id:  # 不是本人
        raise HTTPException(status_code=403, detail="无权操作该商品")  # 403
    db.delete(item)   # 删除
    db.commit()       # 提交
```
