```python
"""认证路由。"""  # 模块说明

from fastapi import APIRouter, Depends, HTTPException  # 路由、依赖、异常
from fastapi.security import OAuth2PasswordRequestForm  # 表单登录字段
from sqlalchemy.orm import Session  # 数据库 Session

from app.database import get_db  # 数据库依赖
from app.models import User      # 用户模型
from app.schemas import Token    # 登录响应模型
from app.security import create_access_token, verify_password  # 签发 token 和校验密码

router = APIRouter(prefix="/auth", tags=["认证"])  # 创建认证路由，统一前缀 /auth


@router.post(  # 注册 POST 路由
    "/login",                 # 登录路径
    response_model=Token,     # 响应模型
    summary="登录并获取JWT",   # Swagger 标题
    description="使用表单提交 username 和 password；密码正确返回 access_token。",  # 说明
)
def login(  # 登录处理函数
    form_data: OAuth2PasswordRequestForm = Depends(),  # 自动解析表单 username/password
    db: Session = Depends(get_db),                     # 数据库依赖
):
    """校验密码并签发 JWT。"""  # 函数说明
    user = db.query(User).filter(User.username == form_data.username).first()  # 按用户名查用户
    if user is None or not verify_password(form_data.password, user.password):  # 用户不存在或密码错误
        raise HTTPException(status_code=401, detail="用户名或密码错误")  # 返回 401
    token = create_access_token(user.id)  # 签发 token
    return Token(access_token=token)      # 返回 token
```
