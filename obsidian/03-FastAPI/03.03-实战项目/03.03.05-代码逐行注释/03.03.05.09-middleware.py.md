```python
"""FastAPI 中间件注册。"""  # 模块说明

import logging  # 日志模块
import time     # 计时模块
import uuid     # 生成 Request-ID

from fastapi import Request  # 请求对象
from fastapi.middleware.cors import CORSMiddleware  # CORS 中间件
from starlette.middleware.base import BaseHTTPMiddleware  # 自定义中间件基类

from app.config import CORS_ORIGINS  # 允许跨域的来源

logger = logging.getLogger(__name__)  # 当前模块日志器


class RequestIDMiddleware(BaseHTTPMiddleware):  # Request-ID 中间件
    async def dispatch(self, request: Request, call_next):  # 处理每个请求
        request_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())  # 用传入 ID 或生成新 ID
        request.state.request_id = request_id  # 把 ID 存到请求状态，方便日志使用
        response = await call_next(request)    # 交给下一个中间件或路由
        response.headers["X-Request-ID"] = request_id  # 响应头带上 Request-ID
        return response  # 返回响应


class LogMiddleware(BaseHTTPMiddleware):  # 日志中间件
    async def dispatch(self, request: Request, call_next):  # 处理每个请求
        start = time.time()  # 记录开始时间
        response = await call_next(request)  # 执行后续处理
        cost = round((time.time() - start) * 1000, 2)  # 计算耗时（毫秒）
        response.headers["X-Process-Time"] = str(cost)  # 响应头返回耗时
        request_id = getattr(request.state, "request_id", "-")  # 读取 Request-ID
        logger.info(  # 输出访问日志
            "[%s] %s %s 耗时 %.2fms",  # 日志格式
            request_id,                # Request-ID
            request.method,            # 请求方法
            request.url.path,          # 请求路径
            cost,                      # 耗时
        )
        return response  # 返回响应


def register_middlewares(app):  # 注册所有中间件
    app.add_middleware(  # 添加 CORS
        CORSMiddleware,  # 中间件类型
        allow_origins=CORS_ORIGINS,  # 允许的来源
        allow_credentials=True,      # 允许携带凭证
        allow_methods=["*"],         # 允许所有方法
        allow_headers=["*"],         # 允许所有请求头
    )
    app.add_middleware(RequestIDMiddleware)  # 添加 Request-ID
    app.add_middleware(LogMiddleware)        # 添加日志
```
