```python
from contextlib import asynccontextmanager  # 导入异步上下文管理器装饰器，用于定义应用生命周期
from pathlib import Path                    # 导入 Path，用于安全拼接文件路径

from fastapi import FastAPI                 # 导入 FastAPI 类，创建应用实例
from fastapi.staticfiles import StaticFiles # 导入 StaticFiles，用于挂载静态文件目录

from app.database import Base, engine                        # 导入 ORM 基类和数据库引擎
from app.exception_handlers import register_exception_handlers  # 导入统一异常处理注册函数
from app.middleware import register_middlewares              # 导入中间件注册函数
from app.routers import auth, files, items, users            # 导入四个业务路由模块

STATIC_DIR = Path(__file__).resolve().parent / "app" / "static"  # 计算静态文件目录的绝对路径


@asynccontextmanager  # 把下面的函数变成异步上下文管理器
async def lifespan(app: FastAPI):  # 定义应用启动/关闭时的生命周期逻辑
    Base.metadata.create_all(bind=engine)  # 启动时自动创建所有数据库表
    yield                                  # 暂停在这里，应用进入运行状态


# 创建 FastAPI 实例
app = FastAPI(title="FastAPI 学习项目", lifespan=lifespan)  # 创建应用，并挂载 lifespan

register_middlewares(app)        # 注册 CORS、Request-ID、日志中间件
register_exception_handlers(app) # 注册统一异常处理器

app.include_router(auth.router)   # 挂载认证路由
app.include_router(users.router)  # 挂载用户路由
app.include_router(items.router)  # 挂载商品路由
app.include_router(files.router)  # 挂载文件/响应示例路由

app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")  # 把静态目录挂载到 /static


@app.get(  # 注册 GET / 路由
    "/",                          # 路由路径
    summary="首页",                # Swagger 里的标题
    description="返回 Hello World 基础示例。",  # Swagger 里的详细说明
)
async def root():                 # 首页处理函数
    """首页接口。"""               # 函数 docstring
    return {"message": "Hello World"}  # 返回 JSON 响应


@app.get(       # 注册 GET /hello/{name} 路由
    "/hello/{name}",  # 路径参数 name
    summary="问候接口", # Swagger 标题
    description="路径参数 name 示例。",  # Swagger 说明
)
async def say_hello(name: str):   # 接收路径参数 name
    """返回问候语。"""             # 函数说明
    return {"message": f"Hello {name}"}  # 返回拼接后的问候语
```
