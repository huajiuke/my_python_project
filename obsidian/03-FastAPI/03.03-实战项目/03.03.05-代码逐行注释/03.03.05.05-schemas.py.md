```python
"""FastAPI Pydantic 请求/响应模型。"""  # 模块说明

from datetime import datetime          # 时间类型

from pydantic import BaseModel, Field  # Pydantic 模型基类和字段校验


class UserRegister(BaseModel):         # 注册请求体
    username: str = Field(..., min_length=3, max_length=20, description="用户名")  # 用户名 3-20 位
    password: str = Field(..., min_length=6, max_length=72, description="密码")    # 密码 6-72 位
    age: int = Field(18, ge=0, le=150, description="年龄")  # 年龄范围
    email: str | None = Field(None, pattern=r"^\S+@\S+\.\S+$", description="邮箱")  # 邮箱格式


class UserOut(BaseModel):   # 用户响应体
    id: int                 # 用户 id
    username: str           # 用户名
    age: int                # 年龄
    email: str | None = None  # 邮箱，可空
    created_at: datetime    # 创建时间


class Token(BaseModel):         # 登录响应体
    access_token: str           # JWT token
    token_type: str = "bearer"  # token 使用方式


class ItemCreate(BaseModel):               # 创建商品请求体
    name: str = Field(..., min_length=1, max_length=50, description="商品名称")  # 商品名
    price: float = Field(..., ge=0, description="商品价格")  # 价格不能为负


class ItemUpdate(BaseModel):                             # 更新商品请求体
    name: str | None = Field(None, min_length=1, max_length=50, description="商品名称")  # 可选名称
    price: float | None = Field(None, ge=0, description="商品价格")  # 可选价格


class ItemOut(BaseModel):  # 商品响应体
    id: int                # 商品 id
    name: str              # 商品名称
    price: float           # 商品价格
    user_id: int           # 所属用户 id
    created_at: datetime   # 创建时间
```
