```python
"""第 6 周 - 第 3 步：FastAPI + SQLAlchemy ORM 模型"""  # 模块说明

from app.database import Base                  # 导入 ORM 基类
from sqlalchemy.orm import Mapped, mapped_column, relationship  # 导入 ORM 类型和关系
from typing import List, Optional             # 类型提示：列表和可选
from datetime import datetime                 # 时间类型
from sqlalchemy import DateTime, Float, ForeignKey, Integer, String  # 数据库列类型


# 用户表
class User(Base):                 # 用户 ORM 模型
    __tablename__ = "users"       # 对应数据库表名

    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)  # 主键，自增
    username: Mapped[str] = mapped_column(String(50), unique=True, index=True, nullable=False)  # 用户名，唯一且索引
    password: Mapped[str] = mapped_column(String(100), nullable=False)  # 密码哈希
    age: Mapped[int] = mapped_column(Integer, default=18)  # 年龄，默认 18
    email: Mapped[Optional[str]] = mapped_column(String(120), nullable=True)  # 邮箱，可空
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)  # 创建时间

    # 一个用户拥有多个商品
    items: Mapped[List["Item"]] = relationship(back_populates="user")  # 一对多关系

    def __repr__(self):  # 便于调试的输出
        return f"<User(id={self.id}, username={self.username!r})>"  # 返回可读表示


# 商品表
class Item(Base):                  # 商品 ORM 模型
    __tablename__ = "items"        # 对应数据库表名

    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)  # 主键，自增
    name: Mapped[str] = mapped_column(String(50), nullable=False)  # 商品名称
    price: Mapped[float] = mapped_column(Float, nullable=False)    # 商品价格
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id"), nullable=False)  # 外键指向用户
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)  # 创建时间

    # 商品属于哪个用户
    user: Mapped["User"] = relationship(back_populates="items")  # 反向关系

    def __repr__(self):  # 便于调试的输出
        return f"<Item(id={self.id}, name={self.name!r})>"  # 返回可读表示
```
